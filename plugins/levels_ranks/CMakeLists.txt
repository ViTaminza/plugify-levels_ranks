# levels_ranks — Plugify C++ plugin: the Levels Ranks core (experience, ranks, statistics, menus)
# and the ILRApi replacement for the eight Pisex LR modules.
#
# Pure s2sdk + plgdb (docs/PLUGIN_STANDARD.md §7): no pisexcompat and no guard library. The map
# gate, client/handle checks and the map-cached schema offsets live in src/lr_gate.cpp; the native
# s2sdk menu lifecycle in src/lr_menus.cpp; the console/chat command layer in src/lr_commands.cpp;
# configs and phrases are read by the plugin-local KV1 reader (src/lr_kv1.cpp).
#
#   lr_core               engine-independent core (settings model, rank/exp/statistics logic, SQL)
#                         also used by tests/lr_core_tests.cpp and tests/lr_bench.cpp
#   lr_support            the engine-facing safety/menu/command/config modules, compiled into the
#                         plugin and into the runtime/config tests
#   lr_bindings           binding globals (__levels_ranks_*) generated by bin/plugify-gen, compiled
#                         into every *consumer* plugin (the LR modules) the same way s2sdk_bindings /
#                         permissions_bindings are
#   plugify-levels_ranks  the plugin shared library (entry "bin/levels_ranks")

set(LEVELS_RANKS_VERSION "1.3.0")

# --- generated consumer bindings ----------------------------------------------
# The .pplugin for the generator needs the version substituted at configure time.
set(LEVELS_RANKS_PPLUGIN "${CMAKE_CURRENT_BINARY_DIR}/levels_ranks.pplugin")
configure_file(
        "${CMAKE_CURRENT_SOURCE_DIR}/levels_ranks.pplugin.in"
        "${LEVELS_RANKS_PPLUGIN}"
)

# plugify-gen writes <output>/external/plugify/include/<name>/...; include/external/plugify/include
# is therefore the include root of the generated consumer API and include/lr_api.hpp (hand written)
# sits next to it.
set(LEVELS_RANKS_INCLUDE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/include/external/plugify/include")
set(LEVELS_RANKS_BINDINGS_CPP "${LEVELS_RANKS_INCLUDE_ROOT}/levels_ranks/levels_ranks.cpp")

# Refreshes the bindings with the current manifest. Best effort by design: the generated headers are
# checked in and are the source of truth for compilation, so a host without the generator keeps
# building (the checked-in files stay untouched).
if(EXISTS "${CMAKE_SOURCE_DIR}/bin/plugify-gen")
    add_custom_command(
            OUTPUT "${LEVELS_RANKS_BINDINGS_CPP}"
            COMMAND sh -c "'${CMAKE_SOURCE_DIR}/bin/plugify-gen' -manifest '${LEVELS_RANKS_PPLUGIN}' -output '${CMAKE_CURRENT_SOURCE_DIR}/include' -lang cpp -overwrite || { echo 'plugify-gen failed (ignored, keeping the checked-in bindings)'; touch '${LEVELS_RANKS_BINDINGS_CPP}'; }"
            DEPENDS "${LEVELS_RANKS_PPLUGIN}"
            COMMENT "plugify-gen: levels_ranks bindings"
            VERBATIM
    )
    add_custom_target(levels_ranks_gen ALL
            DEPENDS "${LEVELS_RANKS_BINDINGS_CPP}")
else()
    add_custom_target(levels_ranks_gen ALL
            COMMAND ${CMAKE_COMMAND} -E echo "bin/plugify-gen not found: keeping the checked-in levels_ranks bindings")
endif()

# The generated binding globals are third-party code: no -Werror on them. Consumers link this
# object library ($<TARGET_OBJECTS:lr_bindings>) exactly like s2sdk_bindings.
set(LR_BINDINGS_OBJECTS "")
if(EXISTS "${LEVELS_RANKS_BINDINGS_CPP}")
    add_library(lr_bindings OBJECT "${LEVELS_RANKS_BINDINGS_CPP}")
    target_include_directories(lr_bindings SYSTEM PUBLIC "${LEVELS_RANKS_INCLUDE_ROOT}")
    target_include_directories(lr_bindings SYSTEM PUBLIC "${CMAKE_SOURCE_DIR}/sdk/include")
    target_compile_definitions(lr_bindings PUBLIC _GLIBCXX_USE_CXX11_ABI=1)
    add_dependencies(lr_bindings levels_ranks_gen)
    set(LR_BINDINGS_OBJECTS "$<TARGET_OBJECTS:lr_bindings>")
endif()

# --- binding globals the plugify module fills ---------------------------------
# In the monorepo / the exported standalone root CMakeLists these targets exist. The fallbacks keep
# a hand-made tree (only this plugin + sdk/) buildable.
if(TARGET s2sdk_bindings)
    set(LR_S2SDK_BINDINGS "$<TARGET_OBJECTS:s2sdk_bindings>")
else()
    add_library(levels_ranks_s2sdk_bindings OBJECT "${CMAKE_SOURCE_DIR}/sdk/include/s2sdk/s2sdk.cpp")
    target_include_directories(levels_ranks_s2sdk_bindings SYSTEM PUBLIC "${CMAKE_SOURCE_DIR}/sdk/include")
    target_compile_definitions(levels_ranks_s2sdk_bindings PUBLIC _GLIBCXX_USE_CXX11_ABI=1)
    set(LR_S2SDK_BINDINGS "$<TARGET_OBJECTS:levels_ranks_s2sdk_bindings>")
endif()
if(TARGET permissions_bindings)
    set(LR_PERMISSIONS_BINDINGS "$<TARGET_OBJECTS:permissions_bindings>")
else()
    add_library(levels_ranks_permissions_bindings OBJECT
            "${CMAKE_SOURCE_DIR}/sdk/include/permissions/permissions.cpp")
    target_include_directories(levels_ranks_permissions_bindings SYSTEM PUBLIC "${CMAKE_SOURCE_DIR}/sdk/include")
    target_compile_definitions(levels_ranks_permissions_bindings PUBLIC _GLIBCXX_USE_CXX11_ABI=1)
    set(LR_PERMISSIONS_BINDINGS "$<TARGET_OBJECTS:levels_ranks_permissions_bindings>")
endif()
if(TARGET hudmenu_bindings)
    set(LR_HUDMENU_BINDINGS "$<TARGET_OBJECTS:hudmenu_bindings>")
else()
    add_library(levels_ranks_hudmenu_bindings OBJECT
            "${CMAKE_SOURCE_DIR}/sdk/include/hudmenu/hudmenu.cpp")
    target_include_directories(levels_ranks_hudmenu_bindings SYSTEM PUBLIC "${CMAKE_SOURCE_DIR}/sdk/include")
    target_compile_definitions(levels_ranks_hudmenu_bindings PUBLIC _GLIBCXX_USE_CXX11_ABI=1)
    set(LR_HUDMENU_BINDINGS "$<TARGET_OBJECTS:levels_ranks_hudmenu_bindings>")
endif()

# --- core ---------------------------------------------------------------------
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/lr_core.cpp")
    add_library(lr_core STATIC src/lr_core.cpp)
    target_include_directories(lr_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(lr_core PUBLIC plgdb)
    target_compile_options(lr_core PRIVATE -Wall -Wextra -Werror)
    target_compile_definitions(lr_core PUBLIC _GLIBCXX_USE_CXX11_ABI=1)
endif()

# --- the config reader and the engine-facing support modules ----------------
# lr_config is also linked by the engine-free core tests (they parse the shipped settings.ini);
# lr_support adds the map gate, the native menu layer and the command layer.
add_library(lr_config STATIC
        src/lr_kv1.cpp
        src/lr_paths.cpp
        src/lr_phrases.cpp
)
target_include_directories(lr_config PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_include_directories(lr_config SYSTEM PRIVATE "${CMAKE_SOURCE_DIR}/sdk/include")
if(TARGET plugify_sdk)
    target_link_libraries(lr_config PUBLIC plugify_sdk)
else()
    target_include_directories(lr_config SYSTEM PUBLIC "${CMAKE_SOURCE_DIR}/sdk/include")
endif()
target_compile_options(lr_config PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(lr_config PUBLIC _GLIBCXX_USE_CXX11_ABI=1)

add_library(lr_support STATIC
        src/lr_gate.cpp
        src/lr_menus.cpp
        src/lr_commands.cpp
)
target_include_directories(lr_support PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
# SYSTEM: warnings from the generated s2sdk/plg/hudmenu headers must not fail -Werror.
target_include_directories(lr_support SYSTEM PRIVATE "${CMAKE_SOURCE_DIR}/sdk/include")
target_link_libraries(lr_support PUBLIC lr_config)
target_compile_options(lr_support PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(lr_support PUBLIC _GLIBCXX_USE_CXX11_ABI=1)

# --- plugin -------------------------------------------------------------------
if(TARGET lr_core AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/plugin.cpp")
    add_library(plugify-levels_ranks SHARED
            src/plugin.cpp
            src/commands.cpp
            src/menus.cpp
            ${LR_S2SDK_BINDINGS}
            ${LR_PERMISSIONS_BINDINGS}
            ${LR_HUDMENU_BINDINGS}
    )
    target_include_directories(plugify-levels_ranks PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    # SYSTEM: warnings from the generated s2sdk/plg/permissions/hudmenu headers must not fail
    # -Werror on our sources.
    target_include_directories(plugify-levels_ranks SYSTEM PRIVATE
            "${CMAKE_SOURCE_DIR}/sdk/include"
            "${LEVELS_RANKS_INCLUDE_ROOT}")
    target_link_libraries(plugify-levels_ranks PRIVATE
            lr_core
            lr_support
            -static-libstdc++ -static-libgcc
    )
    target_compile_options(plugify-levels_ranks PRIVATE -Wall -Wextra -Werror)
    target_compile_definitions(plugify-levels_ranks PRIVATE
            _GLIBCXX_USE_CXX11_ABI=1
            PLUGIFY_MODULE="levels_ranks"
            PLUGIFY_IS_DEBUG=0
            PLUGIFY_IS_RELEASE=1
    )
    set_target_properties(plugify-levels_ranks PROPERTIES OUTPUT_NAME levels_ranks)

    if(UNIX AND NOT APPLE)
        target_link_options(plugify-levels_ranks PRIVATE
                "-Wl,--version-script,${CMAKE_CURRENT_SOURCE_DIR}/sym/version_script.lds")
    endif()
endif()
